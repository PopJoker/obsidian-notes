        private void UpdateAlarmGrid()
        {
            if (!_latestValues.ContainsKey("WarningInf")) return;

            if (dgvAlarms.InvokeRequired)
            {
                dgvAlarms.Invoke(new Action(UpdateAlarmGrid));
                return;
            }

            var alarmData = _latestValues["WarningInf"];
            dgvAlarms.SuspendLayout();

            foreach (var item in alarmData)
            {
                string alarmName = item.Key;
                string status = item.Value;

                // 因為已經預填了，這裡直接找對應的 Row
                foreach (DataGridViewRow row in dgvAlarms.Rows)
                {
                    if (row.Cells["AlarmName"].Value?.ToString() == alarmName)
                    {
                        // 檢查狀態是否有變，有變才更新顏色（優化效能）
                        if (row.Cells["Status"].Value?.ToString() != status)
                        {
                            row.Cells["Status"].Value = status;
                            if (status == "異常")
                            {
                                row.DefaultCellStyle.BackColor = Color.MistyRose;
                                row.DefaultCellStyle.ForeColor = Color.Red;
                            }
                            else
                            {
                                row.DefaultCellStyle.BackColor = Color.White;
                                row.DefaultCellStyle.ForeColor = Color.Black;
                            }
                        }
                        break; // 找到就跳出內層迴圈
                    }
                }
            }
            dgvAlarms.ResumeLayout();
        }
        private void LoadProtocol()
        {
            try
            {
                string path = Path.Combine(Application.StartupPath, "protocol.json");
                if (File.Exists(path))
                {
                    string json = File.ReadAllText(path);
                    _currentProtocol = JsonConvert.DeserializeObject<ProtocolConfig>(json);

                    _latestValues.Clear();
                    string[] sources = { "Grid", "Generator", "Bus", "Load", "PV", "Other", "StateInf", "WarningInf" };
                    foreach (var s in sources) _latestValues[s] = new Dictionary<string, string>();

                    // 1. 初始化 AnalogInf (數值類)
                    if (_currentProtocol.AnalogInf != null)
                    {
                        foreach (var analog in _currentProtocol.AnalogInf)
                        {
                            string sourceKey = GetSourceFromItemName(analog.Name);
                            _latestValues[sourceKey][analog.Name] = $"-- {analog.Unit}";
                        }
                    }

                    // 2. 初始化 StateInf (狀態位類)
                    if (_currentProtocol.StateInf != null)
                    {
                        foreach (var kvp in _currentProtocol.StateInf) // kvp 是 KeyValuePair<string, List<GroupedInfo>>
                        {
                            // 因為 Value 是 List<GroupedInfo>，所以要遍歷裡面的每個 Group
                            foreach (var stateGroup in kvp.Value)
                            {
                                foreach (var bit in stateGroup.Sort) // 現在可以正確存取 Sort 了
                                {
                                    _latestValues["StateInf"][bit.Name] = "讀取中...";
                                }
                            }
                        }
                    }

                    // 3. 初始化 WarningInf (告警位類)
                    if (_currentProtocol.WarningInf != null)
                    {
                        foreach (var kvp in _currentProtocol.WarningInf)
                        {
                            foreach (var warnGroup in kvp.Value)
                            {
                                foreach (var bit in warnGroup.Sort)
                                {
                                    _latestValues["WarningInf"][bit.Name] = "正常";
                                }
                            }
                        }
                    }

                    UpdateDashboardUI();
                    GenerateDynamicUI();
                    InitAlarmDGV();
                    PreFillAlarmGrid();
                    if (_currentProtocol.ConfigInf != null)
                    {
                        GenerateCANConfigUI(_currentProtocol);
                    }
                    listBox_Info.Items.Add($"[系統] 協議載入成功，已初始化所有欄位(含狀態與告警)。");
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show("載入協議失敗: " + ex.Message);
            }
        }

        private void PreFillAlarmGrid()
        {
            if (_currentProtocol.WarningInf == null) return;

            dgvAlarms.Rows.Clear();
            dgvAlarms.SuspendLayout();

            foreach (var kvp in _currentProtocol.WarningInf)
            {
                string sourceDevice = kvp.Key; // 例如 "BCU", "BMU"
                foreach (var warnGroup in kvp.Value)
                {
                    foreach (var bit in warnGroup.Sort)
                    {
                        int rowIndex = dgvAlarms.Rows.Add();
                        DataGridViewRow row = dgvAlarms.Rows[rowIndex];

                        row.Cells["AlarmName"].Value = bit.Name;
                        row.Cells["Status"].Value = "正常"; // 初始狀態
                        row.Cells["Source"].Value = sourceDevice;

                        // 設定初始顏色
                        row.DefaultCellStyle.BackColor = Color.White;
                        row.DefaultCellStyle.ForeColor = Color.Black;
                    }
                }
            }
            dgvAlarms.ResumeLayout();
        }

        /// <summary>
        /// 自動偵測電腦上目前連接的 USBCAN 裝置
        /// </summary>
        private void AutoDetectDevices()
        {
            comboBox_DevIndex.Items.Clear();
            uint devType = m_arrdevtype[comboBox_devtype.SelectedIndex];
            bool found = false;

            // 嘗試偵測 Index 0 到 3 (通常一台電腦不會插超過 4 台)
            for (uint i = 0; i < 4; i++)
            {
                // 嘗試開啟裝置
                if (VCI_OpenDevice(devType, i, 0) == 1)
                {
                    comboBox_DevIndex.Items.Add(i.ToString());
                    VCI_CloseDevice(devType, i); // 偵測完記得先關閉，以免後續正式連線失敗
                    found = true;
                }
            }

            if (found)
            {
                comboBox_DevIndex.SelectedIndex = 0; // 選中第一個偵測到的裝置
            }
            else
            {
                // 如果沒偵測到，補一個 0 進去防止選單空白，並提醒使用者
                comboBox_DevIndex.Items.Add("0");
                comboBox_DevIndex.SelectedIndex = 0;
                MessageBox.Show("未偵測到相容的 CAN 裝置，請檢查硬體連線。", "提示");
            }
        }

        private void BtnWrite_Click(object sender, EventArgs e)
        {
            Button btn = (Button)sender;

            foreach (Control ctrl in tlp.Controls)
            {
                // 只處理輸入控制項
                if (ctrl is NumericUpDown || ctrl is ComboBox)

                    if (ctrl is NumericUpDown nup)

                    }
                    else // ComboBox

                        valueToTransmit = ((ComboBox)ctrl).SelectedIndex;

                    ExecuteCanCommand(param, valueToTransmit);
                }
            }

        private void ExecuteCanCommand(ParameterItem param, double value)
        {
            if (m_bOpen == 0) return;

            VCI_CAN_OBJ sendObj = new VCI_CAN_OBJ();
            // 這裡通常使用擴展幀 (ExternFlag = 1)，並根據協議定義 ID
            sendObj.ID = 0x1801F401; // 範例：機櫃控制專用 CAN ID
            sendObj.SendType = 0;
            sendObj.RemoteFlag = 0;
            sendObj.ExternFlag = 1;
            sendObj.DataLen = 8;

            unsafe
            {
                // 1. 處理數值縮放 (Scale)
                // 若 Scale 為 0.1，則 value 55.5 / 0.1 = 555
                double scaledValue = (param.Scale != 0) ? (value / param.Scale) : value;
                short finalValue = (short)scaledValue;

                // 2. 依照協議格式填充 Data (此處為示意，需對應你的協議手冊)
                sendObj.Data[0] = (byte)param.id;      // 參數索引
                sendObj.Data[1] = 0x06;                // 功能碼：寫入單個寄存器

                // 數值拆分 (Big Endian 範例)
                sendObj.Data[2] = (byte)(finalValue >> 8);
                sendObj.Data[3] = (byte)(finalValue & 0xFF);
            }

            // 發送至控制卡
            VCI_Transmit(m_devtype, m_devind, m_canind, ref sendObj, 1);
        }

        private List<ParameterItem> GetParametersByCategory(ProtocolConfig config, string category)

        private List<ParameterItem> GetParametersByCategory(ConfigRoot config, string category)
        {
            // 這裡使用簡單的 switch 判斷，需與您 JSON 中的分類名稱對應
            switch (category)
            {
                case "控制參數":
                case "控制参数":
                    return config.ConfigInf.控制参数;
                case "通訊參數":
                case "通讯参数":
                    return config.ConfigInf.通讯参数;
                default:
                    return new List<ParameterItem>();
            }
        }

        private void buttonConnect_Click(object sender, EventArgs e)
        {
            if (m_bOpen == 1)
            {
                VCI_CloseDevice(m_devtype, m_devind);
                m_bOpen = 0;
            }
            else

                config.AccCode = Convert.ToUInt32(textBox_AccCode.Text, 16);
                config.AccMask = Convert.ToUInt32(textBox_AccMask.Text, 16);
                config.Timing0 = Convert.ToByte(textBox_Time0.Text, 16);
                config.Timing1 = Convert.ToByte(textBox_Time1.Text, 16);
                config.Filter = (byte)(comboBox_Filter.SelectedIndex + 1);
                config.Mode = (byte)comboBox_Mode.SelectedIndex;
                VCI_InitCAN(m_devtype, m_devind, m_canind, ref config);
                VCI_StartCAN(m_devtype, m_devind, m_canind);
            }
            buttonConnect.Text = m_bOpen == 1 ? "中斷連接" : "連接設備";
            timer_rec.Enabled = m_bOpen == 1;
        }

        unsafe private void timer_rec_Tick(object sender, EventArgs e)
        {
            UInt32 res = VCI_Receive(m_devtype, m_devind, m_canind, ref m_recobj[0], 1000, 100);
            if (res == 0xFFFFFFFF || res == 0) return;

            listBox_Info.BeginUpdate();

            for (UInt32 i = 0; i < res; i++)
            {
                uint id = m_recobj[i].ID;

                if (_currentProtocol != null)
                {
                    // 1. 解析 AnalogInf (數值數據)
                    if (_currentProtocol.AnalogInf != null)
                    {
                        foreach (var analog in _currentProtocol.AnalogInf)
                        {
                            if (id != analog.ServiceCode) continue;

                            string sKey = GetSourceFromItemName(analog.Name); // 縮短變數名避免衝突
                            if (m_recobj[i].DataLen >= (analog.StartByte + analog.ByteSize))
                            {
                                long rawValue = 0;
                                fixed (byte* pData = m_recobj[i].Data)
                                {
                                    if (analog.ByteSize == 2)
                                        rawValue = (pData[analog.StartByte] << 8) | pData[analog.StartByte + 1];
                                    else if (analog.ByteSize == 1)
                                        rawValue = pData[analog.StartByte];
                                }

                                if (!_latestValues.ContainsKey(sKey)) _latestValues[sKey] = new Dictionary<string, string>();
                                _latestValues[sKey][analog.Name] = $"{rawValue} {analog.Unit}";
                            }
                        }
                    }

                    // 2. 解析 StateInf (狀態資訊)
                    if (_currentProtocol.StateInf != null)
                    {
                        foreach (var kvp in _currentProtocol.StateInf)
                        {
                            foreach (var stateGroup in kvp.Value)
                            {
                                if (id != (uint)stateGroup.ServiceCode) continue;

                                // 修正點 1：檢查 DataLen 是否足夠容納 2 個 Byte (因為有 StartBit 8~15)
                                if (m_recobj[i].DataLen >= stateGroup.StartByte + 2)
                                {
                                    // 修正點 2：將連續的兩個 Byte 結合成一個 ushort (16位元)
                                    // 假設是 Little Endian (低位在前)，若順序相反則需調整
                                    ushort fullData = BitConverter.ToUInt16(new byte[] {
                                        m_recobj[i].Data[stateGroup.StartByte],
                                        m_recobj[i].Data[stateGroup.StartByte + 1]
                                    }, 0);

                                    if (!_latestValues.ContainsKey("StateInf"))
                                        _latestValues["StateInf"] = new Dictionary<string, string>();

                                    foreach (var bitInfo in stateGroup.Sort)
                                    {
                                        // 修正點 3：使用 fullData (ushort) 進行位移運算
                                        int bitValue = (fullData >> bitInfo.StartBit) & 0x01;

                                        // 修正點 4：動態讀取 JSON 裡的 "0" 和 "1" 定義 (例如 "斷開"/"閉合")
                                        // 這樣你的顯示文字才會跟著 JSON 走，而不是寫死 "開啟"/"關閉"
                                        string statusText = bitValue == 1 ? bitInfo.One : bitInfo.Zero;

                                        _latestValues["StateInf"][bitInfo.Name] = statusText;
                                    }
                                }
                            }
                        }
                    }

                    // 3. 解析 WarningInf (告警資訊)
                    // 修正點：將所有告警統一存入 "WarningInf" 這個 Key，方便 Grid 更新
                    if (_currentProtocol.WarningInf != null)
                    {
                        foreach (var kvp in _currentProtocol.WarningInf)
                        {
                            string sourceDevice = kvp.Key; // 例如 "BCU"
                            foreach (var warnGroup in kvp.Value)
                            {
                                if (id != (uint)warnGroup.ServiceCode) continue;
                                if (m_recobj[i].DataLen > warnGroup.StartByte)
                                {
                                    byte rawData = m_recobj[i].Data[warnGroup.StartByte];
                                    if (!_latestValues.ContainsKey("WarningInf")) _latestValues["WarningInf"] = new Dictionary<string, string>();

                                    foreach (var bitInfo in warnGroup.Sort)
                                    {
                                        bool isAlarm = ((rawData >> bitInfo.StartBit) & 0x01) == 1;
                                        // 儲存格式可以帶入來源，例如 "BCU-過壓告警"
                                        _latestValues["WarningInf"][bitInfo.Name] = isAlarm ? "異常" : "正常";
                                    }
                                }
                            }
                        }
                    }
                }

                // --- Log 顯示 ---
                string hexId = id.ToString("X");
                string logStr = $"[ID:0x{hexId}] Data: ";
                fixed (byte* pData = m_recobj[i].Data)
                {
                    for (int j = 0; j < m_recobj[i].DataLen; j++)
                        logStr += pData[j].ToString("X2") + " ";
                }
                listBox_Info.Items.Add(logStr);
            }

            while (listBox_Info.Items.Count > 50) listBox_Info.Items.RemoveAt(0);
            listBox_Info.SelectedIndex = listBox_Info.Items.Count - 1;
            listBox_Info.EndUpdate();

            UpdateDashboardUI();
            UpdateAlarmGrid();
        }

            if (name.Contains("母线") || name.Contains("Bus")) return "Bus";
            if (name.Contains("负载") || name.Contains("Load")) return "Load";

        }

        private void UpdateDashboardUI()
        {
            // 使用 BeginInvoke 確保執行緒安全 (如果您未來改用多執行緒接收)
            this.BeginInvoke(new Action(() => {
                foreach (var sourceGroup in _latestValues)
                {
                    foreach (var kvp in sourceGroup.Value)
                    {
                        string dataName = kvp.Key;   // 數據名稱，例如 "電池電壓"
                        string valStr = kvp.Value;    // 數值，例如 "54.2 V"

                        if (_controlCache.ContainsKey(dataName))
                        {
                            _controlCache[dataName].Text = $"{dataName}: {valStr}";
                        }
                    }
                }
            }));
        }

        private void UpdateLabelText(Label lbl, string sourceKey)
        {

                }

            }
        }

        {

            unsafe
            {
                VCI_CAN_OBJ sendobj = new VCI_CAN_OBJ();
                sendobj.RemoteFlag = (byte)comboBox_FrameFormat.SelectedIndex;
                sendobj.ExternFlag = (byte)comboBox_FrameType.SelectedIndex;
                sendobj.ID = Convert.ToUInt32(textBox_ID.Text, 16);

                string[] dataBytes = textBox_Data.Text.Trim().Split(' ');
                sendobj.DataLen = (byte)Math.Min(dataBytes.Length, 8);

                byte* pData = sendobj.Data;
                for (int i = 0; i < sendobj.DataLen; i++)
                {
                    pData[i] = Convert.ToByte(dataBytes[i], 16);
                }

                if (VCI_Transmit(m_devtype, m_devind, m_canind, ref sendobj, 1) == 0)
                    MessageBox.Show("發送失敗");
            }
